UPDATE global_nike 
SET gender_clean = CASE 
    
    -- 2. Exact single categories
    WHEN gender_segment = 'MEN' THEN 'Men'
    WHEN gender_segment = 'WOMEN' THEN 'Women'
    
    -- 3. Handle combinations with both Men and Women -> Unisex
    WHEN gender_segment LIKE '%MEN%' AND gender_segment LIKE '%WOMEN%' THEN 'Unisex'
    
    -- 4. Handle Kids (Boys, Girls, or combinations like Women|Girls)
    WHEN gender_segment LIKE '%BOYS%' OR gender_segment LIKE '%GIRLS%' THEN 'Kids'
    
    -- 5. Fallback for any other mixed or unhandled values
    ELSE gender_segment
END;


UPDATE global_nike
SET gender_clean =
    CASE

        WHEN LOWER(subcategory) LIKE '%kid%' 
          OR LOWER(subcategory) LIKE '%junior%' 
          OR LOWER(subcategory) LIKE '%baby%'
          OR subcategory LIKE '%キッズ%' 
          OR subcategory LIKE '%ジュニア%' 
          OR subcategory LIKE '%リトルキッズ%' THEN 'Kids'

        WHEN LOWER(subcategory) LIKE '%women%' 
          OR LOWER(subcategory) LIKE '%woman%' 
          OR subcategory LIKE '%ウィメンズ%' THEN 'Women'

        WHEN LOWER(subcategory) LIKE '%men%' 
          OR LOWER(subcategory) LIKE '%man%' 
          OR subcategory LIKE '%メンズ%' THEN 'Men'

        ELSE 'Unisex'
    END
WHERE gender_clean IS NULL OR TRIM(gender_clean) = '';

ALTER TABLE global_nike ADD COLUMN price_usd REAL;

UPDATE global_nike
SET price_usd = ROUND(
    CASE 
        WHEN currency = 'USD' THEN price_local
        WHEN currency = 'GBP' THEN price_local * 1.30
        WHEN currency = 'EUR' THEN price_local * 1.09
        WHEN currency = 'JPY' THEN price_local * 0.0068
        ELSE price_local
    END, 2);


SELECT 
    country_code,
    ROUND(AVG(price_usd), 2) AS avg_price_usd,
    COUNT(*) AS total_products
FROM global_nike
GROUP BY country_code
ORDER BY avg_price_usd DESC;


ALTER TABLE global_nike ADD COLUMN country_total_value REAL;
ALTER TABLE global_nike ADD COLUMN country_total_products INTEGER;


UPDATE global_nike
SET 
    country_total_value = stats.total_val,
    country_total_products = stats.total_prods
FROM (
    SELECT 
        country_code,
        ROUND(SUM(price_usd), 2) AS total_val,
        COUNT(*) AS total_prods
    FROM global_nike
    GROUP BY country_code
) AS stats
WHERE global_nike.country_code = stats.country_code;
